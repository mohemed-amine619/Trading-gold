# requirements-core.txt — Linux / Docker dependencies for the Ubuntu core.
#
# This file is the ONLY pip target used by the Docker image. It deliberately
# EXCLUDES MetaTrader5, which is a Windows-only package with no Linux wheel —
# including it here makes `pip install -r` fail outright on Ubuntu.
#
#   Ubuntu core (this file)          Windows MT5 bridge (requirements-mt5.txt)
#   ------------------------         ---------------------------------------
#   core/, api/, alembic/            bridge/ (FastAPI + MetaTrader5)
#
# Verified against Python 3.12. Pin exact versions for reproducible builds.

# ── Web / API ────────────────────────────────────────────────────────────────

uvicorn[standard]==0.54.0

# ── Settings & validation ────────────────────────────────────────────────────

python-dotenv==1.2.4

# ── Database ─────────────────────────────────────────────────────────────────

# transitive dependency that pip may or may not resolve depending on the
# resolver, so it is pinned explicitly — a missing greenlet produces a
# confusing ImportError at `import sqlalchemy.ext.asyncio`, not at pip time.
SQLAlchemy[asyncio]==2.1.4
greenlet==3.5.6            # required by SQLAlchemy's asyncio layer
alembic==1.20.0
